local origem=sala:GetPivot()
local grupos={}
local gavetas={}
for _,movel in ipairs(sala.Moveis:GetChildren()) do
    for _,g in ipairs(movel:GetChildren()) do
        table.insert(grupos,g)
        if g:GetAttribute("Tipo")=="gaveta" then table.insert(gavetas,g) end
    end
end
-- O controlador da Galeria le as coordenadas ao receber a tag. Registrar depois
-- do posicionamento permite usar o mesmo modelo nos quatro giros do gerador.
for _,g in ipairs(grupos) do
    local zona=origem*g:GetAttribute("ZonaLocal")
    if g:GetAttribute("Tipo")=="gaveta" then
        g:SetAttribute("Curso",origem:VectorToWorldSpace(g:GetAttribute("CursoLocal")))
        g:SetAttribute("LevaCF",zona); g:SetAttribute("LevaTam",g:GetAttribute("Tam"))
    else
        g:SetAttribute("Pivo",origem:PointToWorldSpace(g:GetAttribute("PivoLocal")))
        g:SetAttribute("Eixo",origem.UpVector)
        g.Parent:SetAttribute("InteriorCF",zona); g.Parent:SetAttribute("InteriorTam",g:GetAttribute("Tam"))
    end
    tags:AddTag(g,"Invokyr_Abre")
end

local config=folder(room,"ConfigurarPortas","ModuleScript")
config.Source=[==[
-- require(sala.ConfigurarPortas)(sala, {N=100,S=0})
-- Um angulo (0 a 100) habilita a porta; um lado omitido recebe o painel.
return function(room,angles)
    local function show(parent,on)
        for _,p in ipairs(parent:GetDescendants()) do
            if p:IsA("MeshPart") then p.Transparency=on and 0 or 1
            elseif p:IsA("BasePart") and p.Name:match("Colisao$") then
                p.CanCollide=on
                p.CanQuery=on
            end
        end
    end
    for _,side in ipairs(room.Portas:GetChildren()) do
        local angle=angles[side.Name]
        local active=type(angle)=="number"
        show(side.Fecho,not active)
        for _,label in ipairs({"A","B"}) do
            local leaf=side[label]
            show(leaf,active)
            local turn=active and math.clamp(angle,0,100) or 0
            leaf:PivotTo(room:GetPivot()*leaf:GetAttribute("Fechada")*CFrame.Angles(0,math.rad(label=="A" and -turn or turn),0))
        end
    end
end
]==]
