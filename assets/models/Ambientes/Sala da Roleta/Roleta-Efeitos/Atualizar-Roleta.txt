-- Integrate velocity: gradual load, nearly steady travel, longer braking.
local function travel(t)
 local up,down=.26,.37
 local full=1-(up+down)/2
 if t<up then return (t/2-up*math.sin(math.pi*t/up)/(2*math.pi))/full end
 if t<=1-down then return (t-up/2)/full end
 local left=1-t
 return 1-(left/2-down*math.sin(math.pi*left/down)/(2*math.pi))/full
end
local function spin()
 local sector=math.round(angle/stepAngle)%12
 local options={2,3,4,5,7}
 local steps=options[rng:NextInteger(1,#options)]
 local start=sector*stepAngle
 local finish=(sector+steps)%12
 local lastSector=-1
 room:SetAttribute("InicioDoGiro",workspace:GetServerTimeNow())
 for _,s in ipairs({axle,rumble,debris}) do
  if s.SoundId~="" then s:Play() end
 end
 for _,e in ipairs(dust) do e.Enabled=true end
 animate(spinDuration,function(t)
  local a=start+steps*stepAngle*travel(t)
  spinAt(a)
  local crossing=math.round(a/stepAngle)%6
  if crossing~=lastSector then
   tint(colors[crossing+1])
   lastSector=crossing
  end
 end)
 for _,e in ipairs(dust) do e.Enabled=false end
 room:SetAttribute("InicioDoGiro",nil)
 angle=finish*stepAngle
 spinAt(angle)
 return finish%6
end

  if not room:IsDescendantOf(workspace) then return end

   local v=position(p.Character)
   if v then
    if onDeck(v) then
     count+=1
    elseif entrance(v) or (p==requester and nearPortal(v,sign)) then
     move(p.Character,-16+(count%8)*4.3,55.6,16.6-(math.floor(count/8)%9)*3.8,true)

 end)
end
